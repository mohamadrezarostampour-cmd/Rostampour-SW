rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null && request.auth.token.email == 'rostampour@rostampour-system.local';
    }
    function isStation() {
      return request.auth != null && request.auth.token.email == 'station@rostampour-system.local';
    }
    function isSignedIn() {
      return isAdmin() || isStation();
    }
    function stationCanDeleteReadings() {
      return isStation() &&
        get(/databases/$(database)/documents/settings/permissions).data.deleteOwnReadings == true;
    }

    // مشترکین، تعرفه‌ها، انواع کنتور، مسیرهای قرائت: فقط مدیر می‌تواند بنویسد
    match /consumers/{id} {
      allow read: if isSignedIn();
      allow create, update, delete: if isAdmin();
    }
    match /tariffs/{id} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /meterTypes/{id} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /routes/{id} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    // قرائت‌ها: هر دو نقش می‌توانند بخوانند و ثبت کنند؛ ویرایش فقط مدیر؛
    // حذف: مدیر همیشه مجاز، مأمور فقط اگر سوییچ «حذف قرائت‌های خودش» در تنظیمات فعال باشد
    match /readings/{id} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update: if isAdmin();
      allow delete: if isAdmin() || (stationCanDeleteReadings() && resource.data.agentRole == 'station');
    }

    // لاگ سیستم: همه می‌توانند رخداد ثبت کنند، فقط مدیر می‌تواند بخواند/ویرایش کند (برای بازیابی از پشتیبان)
    match /auditLogs/{id} {
      allow create: if isSignedIn();
      allow read, update, delete: if isAdmin();
    }

    // تنظیمات (سطوح دسترسی و تنظیمات عمومی): همه بخوانند، فقط مدیر بنویسد
    match /settings/{id} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    // ================= سامانه CMM (نگهداری و تعمیرات) - مجموعه‌های جدا و مستقل =================
    match /cmm_equipment/{document=**} {
      allow read, write: if request.auth != null;
    }
    match /cmm_pm/{document=**} {
      allow read, write: if request.auth != null;
    }
    match /cmm_cm/{document=**} {
      allow read, write: if request.auth != null;
    }
    match /cmm_tanks/{document=**} {
      allow read, write: if request.auth != null;
    }
    match /cmm_plants/{document=**} {
      allow read, write: if request.auth != null;
    }
    match /cmm_pumping/{document=**} {
      allow read, write: if request.auth != null;
    }
    match /cmm_activitylog/{document=**} {
      allow read, write: if request.auth != null;
    }
    match /cmm_meta/{document=**} {
      allow read, write: if request.auth != null;
    }

    // ================= سامانه انبار (Warehouse) - مجموعه‌ی جدید و مستقل =================
    // نکته: این پروژه‌ی ادغام‌شده کاملاً مستقل از پروژه‌ی مستقل انبار قبلی است و
    // از یک سند جدید (wh_meta/warehouse_data) استفاده می‌کند تا هیچ تداخلی با
    // داده‌های موجود ایجاد نشود.
    match /wh_meta/{document=**} {
      allow read, write: if request.auth != null;
    }

    // هر مسیر تعریف‌نشده دیگر: کاملاً مسدود
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
